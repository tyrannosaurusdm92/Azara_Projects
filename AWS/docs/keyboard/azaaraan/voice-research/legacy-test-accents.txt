/* Run from any directory: node docs/test-accents.js */
'use strict';
const fs=require('fs'),path=require('path'),vm=require('vm');
const root=path.resolve(__dirname,'..');
const json=n=>JSON.parse(fs.readFileSync(path.join(root,'json',n),'utf8'));
const D=json('azaaraan-system.json'),P=json('tts-pronunciation.json');
let faults=0;
function assert(ok,message){if(!ok){console.error('FAIL',message);faults++;}}
const ui={};const handlers={};
const elem=id=>ui[id]||(ui[id]={value:id==='voiceBase'?'100':id==='speechRate'||id==='speechVolume'?'1':'',textContent:'',disabled:false,addEventListener(event,fn){handlers[id+':'+event]=fn;}});
const testVoice={name:'Microsoft Zira Browser Voice',lang:'en-US'};
const synth={utterance:null,getVoices(){return [testVoice];},cancel(){},speak(u){this.utterance=u;}};
const window={speechSynthesis:synth,SpeechSynthesisUtterance:function(text){this.text=text;},addEventListener(){}};
const document={getElementById:elem,querySelectorAll(){return []},querySelector(){return {click(){}}}};
const context={window,document,console,setTimeout(){}};vm.createContext(context);
for(const file of ['js/data.js','js/tts-data.js','js/ipa-data.js','js/ipa.js','js/tts.js'])vm.runInContext(fs.readFileSync(path.join(root,file),'utf8'),context);
assert(JSON.stringify(D.marks)===JSON.stringify(context.window.AZA_DATA.marks),'Offline script dictionary differs from JSON');
assert(JSON.stringify(P)===JSON.stringify(context.window.AZA_TTS_DATA),'Offline speech dictionary differs from JSON');
const tests=[
['T',['C01'],'tæ'],['B',['C02'],'bɛ'],['P',['C03'],'pɪ'],['R',['C04'],'rʌ'],
['K',['C05'],'kɑː'],['S',['C06'],'siː'],['N',['C07'],'nuː'],['N',['C08'],'ŋ'],
['K',['C09'],'kw'],['B',['C09'],'v'],['N',['C10'],'ɲ'],['O',['C10'],'ɔɪ'],
['B',['C11'],'f'],['W',['C12'],'b'],['T',['C01','K01'],'tæz'],['K',['K02'],'ks'],
['T',['K03'],'tʃ'],['D',['K03'],'dʒ'],['T',['K04'],'θ'],['S',['K04'],'ʃ'],
['O',['K04'],'oʊ'],['T',['K05'],'t'],['B',['C02','K06'],'bɛd'],['B',['C01','K07'],'bæk'],
['B',['C02','K08'],'bɛl'],['S',['C01','K09'],'sæm'],['P',['C03','K10'],'pɪn'],
['K',['C01','K11'],'kær'],['K',['C01','K12'],'kæt']
];
const {transcribe,pitchFromBase}=window.AzaaraanTTS;
for(const [base,ids,expected] of tests){
 const char=ids.length&&D.vowelCarriers[base]?D.vowelCarriers[base]:D.keys[base];
 const spelling=char+ids.map(id=>D.marks.find(x=>x.id===id).output).join('');
 assert(transcribe(spelling).ipa==='/'+expected+'/',base+' + '+ids.join(' + ')+' = '+JSON.stringify(transcribe(spelling))+' instead of /'+expected+'/');
}
const extra=[['T́','tæ'],['B̀','bɛ'],['Ḃ','f'],['Aza’raan','æzʌʔr͡ræn'],['Aza’ra','æzʌʔr͡rʌ']];
for(const [input,expected] of extra)assert(transcribe(input).ipa==='/'+expected+'/',input+' canonical/Latin combining IPA');
assert(D.marks.length===28,'Must be 24 accent + 4 special signs');
assert(D.marks.slice(0,24).length===24,'Must have all 24 accent IDs');
assert(ui.voiceBase.value==='100'&&ui.voiceBaseOut.textContent==='Feminine','Initial voice-base at far right');
const fixed=transcribe('ត៊ា');
for(const [pos,label,pitch] of [[0,'Masculine',0.6],[50,'Androgynous',1.1],[100,'Feminine',1.6]]){
 ui.voiceBase.value=String(pos);handlers['voiceBase:input']();
 assert(ui.voiceBaseOut.textContent===label,label+' voice-base output');
 assert(Math.abs(pitchFromBase(pos)-pitch)<1e-6,'Pitch at voice-base '+pos);
 assert(transcribe('ត៊ា').ipa===fixed.ipa,'Pitch changed transcription at '+pos);
 ui.speechText.value='ត៊ា';handlers['speakText:click']();
 assert(synth.utterance.text===fixed.speech,'Pitch changed speech content at '+pos);
 assert(synth.utterance.voice===testVoice,'Pitch changed selected voice');
}
// An imported impossible stack must be diagnosed rather than invented.
const badStack=transcribe(D.keys.T+D.marks[0].output+D.marks[1].output);
assert(badStack.uncertain,'Competing vowel accents should produce a warning');
const unknownHardH=transcribe(D.keys.K+D.marks.find(m=>m.id==='K04').output);
assert(unknownHardH.uncertain,'Undefined Hard H context should produce a warning');
console.log(`${tests.length} creator examples + ${extra.length} Latin/name cases + voice/policy checks: ${faults?'FAILED '+faults:'PASSED'}`);
process.exitCode=faults?1:0;
